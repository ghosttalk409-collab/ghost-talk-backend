function initAdminSocket() { const proto = window.location.protocol === 'https:' ? 'wss://' : 'ws://'; const wsUrl = proto + window.location.host; ws = new WebSocket(wsUrl); ws.onopen = () => { document.getElementById('connectionStatus').innerText = 'AUTHORIZED | ONLINE'; ws.send(JSON.stringify({ type: 'admin-auth', secret: 'GHOST-ADMIN-PASS-2026' })); }; ws.onmessage = (evt) => { const msg = JSON.parse(evt.data); if (msg.type === 'admin-authenticated' && msg.pricing) { document.getElementById('monthlyPriceInput').value = msg.pricing.monthly; document.getElementById('yearlyPriceInput').value = msg.pricing.yearly; } if (msg.type === 'admin-telemetry') updateDashboardUI(msg.metrics); }; ws.onclose = () => { document.getElementById('connectionStatus').innerText = 'OFFLINE | RECONNECTING...'; setTimeout(initAdminSocket, 2000); }; }
